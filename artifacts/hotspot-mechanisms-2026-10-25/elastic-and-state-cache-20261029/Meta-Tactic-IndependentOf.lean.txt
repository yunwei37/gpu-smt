12: open Meta
13: 
14: /--
15: Check if a goal is "independent" of a list of other goals.
16: We say a goal is independent of other goals if assigning a value to it
17: can not change the assignability of the other goals.
18: 
19: Examples:
20: * `?m_1 : Type` is not independent of `?m_2 : ?m_1`,
21:   because we could assign `true : Bool` to `?m_2`,
22:   but if we first assign `Nat` to `?m_1` then that is no longer possible.
23: * `?m_1 : Nat` is not independent of `?m_2 : Fin ?m_1`,
24:   because we could assign `37 : Fin 42` to `?m_2`,
25:   but if we first assign `2` to `?m_1` then that is no longer possible.
26: * `?m_1 : ?m_2` is not independent of `?m_2 : Type`, because we could assign `Bool` to ?m_2`,
27:   but if we first assign `0 : Nat` to `?m_1` then that is no longer possible.
28: * Given `P : Prop` and `f : P → Type`, `?m_1 : P` is independent of `?m_2 : f ?m_1`
29:   by proof irrelevance.
30: * Similarly given `f : Fin 0 → Type`, `?m_1 : Fin 0` is independent of `?m_2 : f ?m_1`,
31:   because `Fin 0` is a subsingleton.
32: * Finally `?m_1 : Nat` is independent of `?m_2 : α`,
33:   as long as `?m_1` does not appear in `Meta.getMVars α`
34:   (note that `Meta.getMVars` follows delayed assignments).
35: 
36: This function only calculates a conservative approximation of this condition.
37: That is, it may return `false` when it should return `true`.
38: (In particular it returns false whenever the type of `g` contains a metavariable,
39: regardless of whether this is related to the metavariables in `L`.)
40: -/
41: def isIndependentOf (L : List MVarId) (g : MVarId) : MetaM Bool := g.withContext do
42:   let t ← instantiateMVars (← g.getType)
43:   if t.hasExprMVar then
44:     -- If the goal's type contains other meta-variables,
45:     -- we conservatively say that `g` is not independent.
46:     -- It would be possible to check if `L` depends on these meta-variables.
47:     return false
48:   if (← inferType t).isProp then
49:     -- If the goal is propositional,
50:     -- proof-irrelevance ensures it is independent of any other goals.
51:     return true
52:   if ← g.isSubsingleton then
53:     -- If the goal is a subsingleton, it is independent of any other goals.
54:     return true
55:   -- Finally, we check if the goal `g` appears in the type of any of the goals `L`.
56:   L.allM fun g' => do pure !((← getMVarDependencies g').contains g)
57: 
58: end Lean.MVarId
